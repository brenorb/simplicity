packV :: T.Vector a b -> [a] -> b
packV T.SingleV [a]=a
packV (T.DoubleV v) xs=(packV v (H.take n xs),packV v (H.drop n xs)) where n=T.vectorSize v
packV _ _=error "fixed vector encoding length"
packT :: (Core t,TyC a) => T.Vector Word8 b -> [t a Word8] -> t a b
packT T.SingleV [a]=a
packT (T.DoubleV v) xs=packT v (H.take n xs)&&&packT v (H.drop n xs) where n=T.vectorSize v
packT _ _=error "fixed parser field length"

quorum :: forall t. Simplicity t => t VInput ()
quorum=(iden&&&vBodyHash)>>>count>>>((konst (w8 13)&&&iden)>>>A.le word8>>>verify)
 where
  count :: t (VInput,Word256) Word8
  count=L.foldl' (\acc i->(acc&&&slot i)>>>A.add word8>>>ih) (konst (w8 0)) [0..18]
  slot i=((take (drop (at32 i)))&&&ih)>>>match (konst (w8 0)) verifyOne

bridgeT :: (Simplicity t,Primitive t) => t VInput ()
bridgeT=allChecks [take (take base),framing,fields,amounts,replay,stateTransition,payout,quorum]
 where
  input t=take (take t)
  framing=allChecks [same (byte i) (konst (w8 0))|i<-[184..255]]
  fields=allChecks [same (p16 8) (konst (w16 2)),same (p256 10) (M.pw emitter),same (byte 50) (konst (w8 32)),same (byte 51) (konst (w8 1)),same (p256 84) (konst (w256 M.nativeAsset)),same (p16 116) (konst (w16 2)),same (p16 150) (konst (w16 destinationChain)),input (same mode (konst (w8 0))),input (same actor (k 0)),allChecks [same (drop (at32 i)>>>isSome>>>not iden) true|i<-[19..31]],input (allChecks [same (o i) (k 0)|i<-[3..14]])]
  quantity=p256 52>>>iih;fee=p256 152>>>iih
  amounts=allChecks [((p256 52&&&konst (w256 (2^64)))>>>A.lt word256>>>verify),((p256 152&&&konst (w256 (2^64)))>>>A.lt word256>>>verify),nonzero fee,lt fee quantity,same quantity (input amount)]
  relative=sub (p64 42) (input (o 2))
  mask=(relative&&&k 1)>>>W.left_shift word64 word64
  bitmap=input (o 1)
  replay=allChecks [lt relative (k 64),same ((bitmap&&&mask)>>>W.bitwise_and word64) (k 0),same (input (n 1)) ((bitmap&&&mask)>>>W.bitwise_or word64)]
  stateTransition=input (changes [(0,sub (o 0) amount),(1,n 1),nonce 15])
  payout=allChecks [input (outCount 3),input (inputFund (k 0)),input continuation,same (input (outScript 1)) (p256 118),same (input (outVal 1)) (sub quantity fee),same (input (outScript 2)) (destination 2),same (input (outVal 2)) fee]
bridgeProgram :: (Simplicity t,Primitive t) => VaaW -> t () ()

bridgeInput :: VaaW -> VInput
bridgeInput (g,body,sigs)=encoded
 where
  padded=if BS.length body==184 then body<>BS.replicate 72 0 else body
  encoded=((encode g,packV vector256 (map (w8 . toInteger) (BS.unpack padded))),packV vector32 (map sigWit (H.take 32 (sigs++repeat Nothing))))
  sigWit Nothing=Left ()
  sigWit (Just (r,(s,p)))=Right (w256 r,(w256 s,if p then Right () else Left ()))
bodyRef seqValue quantity target=runPut (putWord32be 1700000000>>putWord32be 17>>putWord16be 2)<>emitter<>runPut (putWord64be (fromInteger seqValue))<>BS.pack [32,1]<>R.bytesFromInt quantity<>R.bytesFromInt M.nativeAsset<>runPut (putWord16be 2)<>target<>runPut (putWord16be (fromInteger destinationChain))<>R.bytesFromInt 500
vEnv oldState newState quantity cmr=mkEnv transaction 0 (\_->stateTap cmr (stateRef oldState)) cmr
 where
  ins=[mkInput (dummyOutpoint 0) (fromInteger (H.head oldState+carrier)) (BL.fromStrict (stateScript cmr (stateRef oldState))) 0xfffffffe,mkInput (dummyOutpoint 1) 1000 (BL.fromStrict (recipient 9)) 0xfffffffe]
  outs=[mkOutput (fromInteger (H.head newState+carrier)) (stateScript cmr (stateRef newState)),mkOutput (fromInteger (quantity-500)) (recipient 0),mkOutput 500 (recipient 2)]
  transaction=mkTx 2 ins outs 0
bridgeVector name sequenceValue bitmap quantity signedQuantity wrongRecipient numSignatures duplicateGuardian badPadding expected=Vector name env (g,body,signatures) expected
 where
  recipientHash=hashToBytes (sha256 (recipient 0))
  target=if wrongRecipient then hashToBytes (sha256 (recipient 1)) else recipientHash
  rawBody=bodyRef sequenceValue quantity target
  body=if badPadding then rawBody<>BS.singleton 1<>BS.replicate 71 0 else rawBody
  signedBody=bodyRef sequenceValue signedQuantity target
  digest=R.intFromBytes (K.keccak256 (K.keccak256 signedBody))
  sign i=let (r,s,p)=M.ecdsaSign (toInteger i+1) digest (toInteger i+101) in Just (r,(s,p))
  signatures=[if i>=numSignatures then Nothing else sign (if duplicateGuardian&&i==1 then 0 else i)|i<-[0..18]]
  relative=sequenceValue-100
  newBitmap=bitmap H.+ (if relative>=0&&relative<64&&notBit bitmap relative then 2^relative else 0)
  notBit a b=H.not (testBit a (fromInteger b))
  oldState=pad [100000,bitmap,100]
  newState=pad ([100000-quantity,newBitmap,100]++replicate 12 0++[1])
  g=(0,(oldState,newState),(0,quantity),(zeroSig,zeroSig))
  env=vEnv oldState newState quantity
