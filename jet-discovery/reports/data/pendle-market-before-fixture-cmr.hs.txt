validTerm :: Assert t => t Ledger ()
validTerm=allChecks [le (add (field 1) (field 2)) (field 0),same (field 1) (sumT (map field [3,4,7,9])),same (field 2) (sumT (map field [5,6,8])),same (field 11) (sumT (map field [12,13,19])),same (field 20) (k P.year),le (field 16) (k (2*P.year)),le (k P.scale) (field 17),le (field 17) (k (2*P.scale)),allChecks [same (field i) (k 0)|i<-[10,15]++[21..31]],select (is (field 11) (k 0)) empty live]
 where
  empty=allChecks [same (field i) (k 0)|i<-[7,8,18,19]]
  live=allChecks [same (field 19) (k 1000),nonzero (field 7),nonzero (field 8),le (field 7) (k (10^9)),le (field 8) (k (10^9)),nonzero (field 18),le (field 18) (k P.scale)]
initialWrapped :: J.WrappedSimplicity P.Base Word64
initialWrapped=J.jetSubst P.initialRate
{-# NOINLINE initialWrapped #-}
initialLn :: Simplicity t => t P.Base Word64
initialLn=J.unwrap initialWrapped

termT :: (Simplicity t,Primitive t) => CT t
termT=allChecks [base,old>>>validTerm,new>>>validTerm,choose (claims++[(16,event),(17,join),(18,trade True),(19,trade False),(20,exit),(21,pullFee),(22,transferLp)])]
 where
  claims=H.take 5 claimBranches++[(5,allChecks [le (k P.year) (o 16),claimAction True 2])]
  active=lt (o 16) (k P.year)
  remaining=sub (k P.year) (o 16)
  event=allChecks [sigAt 3 0,lt (o 16) amount,le (o 17) (aux 0),changes [(16,amount),(17,aux 0),nonce],noCash]
  quoteJoin=(o 7&&&(o 8&&&(o 11&&&(aux 0&&&aux 1))))>>>Q.joinQuote
  join=(allChecks [actorSig,active]&&&iden)>>>drop ((iden&&&quoteJoin)>>>(iden&&&initialOrOld)>>>joinChecks)
  initialOrOld=select (is (take (o 11)) (k 0)) ((add (take (o 7)) (drop oh)&&&(add (take (o 8)) (drop ioh)&&&(take (o 17)&&&(take remaining&&&k 0))))>>>initialLn) (take (o 18))
  joinChecks=allChecks [le (jt (aux 2)) minted,changesAt jo jn ([(7,add (jo 7) sy),(8,add (jo 8) pt),(11,add (jo 11) (add minted locked)),(12,jAccount (add (jo 12) minted) (jo 12)),(13,jAccount (jo 13) (add (jo 13) minted)),(14,add (jo 14) (k 1)),(18,ih),(19,add (jo 19) locked)]++jSet False (sub (jt (userBalance False o)) sy)++jSet True (sub (jt (userBalance True o)) pt)),jt noCash]
   where
    jt x=take (take x)
    jo i=jt (o i);jn i=jt (n i)
    sy=take (drop oh);pt=take (drop ioh);minted=take (drop iioh);locked=take (drop iiih)
    jAccount a b=select (is (jt actor) (k 0)) a b
    jSet isB value=if isB then [(5,jAccount value (jo 5)),(6,jAccount (jo 6) value)] else [(3,jAccount value (jo 3)),(4,jAccount (jo 4) value)]
  poolBase=o 7&&&(o 8&&&(o 17&&&(remaining&&&o 18)))
  trade buy=(allChecks [actorSig,active,nonzero (o 11),nonzero amount]&&&iden)>>>drop ((iden&&&((poolBase&&&(konst (toBit buy)&&&amount))>>>P.trade))>>>tradeChecks buy)
  tradeChecks buy=allChecks [same (net>>>oh) (konst (toBit buy)),nonzero netMag,limit,changesAt oldP newP ([(7,drop oh),(8,drop ioh),(9,add (oldP 9) reserveFee),(14,add (oldP 14) (k 1)),(18,drop iiih)]++setP False ((if buy then sub else add) (take (userBalance False o)) netMag)++setP True ((if buy then add else sub) (take (userBalance True o)) (take amount))),take noCash]
   where
    q=drop iioh;net=q>>>oh;netMag=net>>>ih;reserveFee=q>>>iih
    oldP i=take (o i);newP i=take (n i)
    limit=select (is (take (aux 0)) (k 0)) unit (if buy then le netMag (take (aux 0)) else le (take (aux 0)) netMag)
    accountP a b=select (is (take actor) (k 0)) a b
    setP isB value=if isB then [(5,accountP value (oldP 5)),(6,accountP (oldP 6) value)] else [(3,accountP value (oldP 3)),(4,accountP (oldP 4) value)]
  exitQuote=G.floorRatio (o 7) amount (o 11)&&&G.floorRatio (o 8) amount (o 11)
  exit=(allChecks [actorSig,nonzero amount,le amount (account (o 12) (o 13))]&&&iden)>>>drop ((iden&&&exitQuote)>>>exitChecks)
  exitChecks=allChecks [nonzero (add ioh iih),le (take (aux 0)) ioh,le (take (aux 1)) iih,changesAt oldP newP ([(7,sub (oldP 7) ioh),(8,sub (oldP 8) iih),(11,sub (oldP 11) (take amount)),(12,accountP (sub (oldP 12) (take amount)) (oldP 12)),(13,accountP (oldP 13) (sub (oldP 13) (take amount))),(14,add (oldP 14) (k 1))]++setP False (add (take (userBalance False o)) ioh)++setP True (add (take (userBalance True o)) iih)),take noCash]
   where
    oldP i=take (o i);newP i=take (n i)
    accountP a b=select (is (take actor) (k 0)) a b
    setP isB value=if isB then [(5,accountP value (oldP 5)),(6,accountP (oldP 6) value)] else [(3,accountP value (oldP 3)),(4,accountP (oldP 4) value)]
  pullFee=allChecks [sigAt 2 0,nonzero amount,konst (w32 800000)>>>TL.checkLockHeight,changes [(0,sub (o 0) amount),(1,sub (o 1) amount),(9,sub (o 9) amount),nonce],withdraw amount (destination 2)]
  transferLp=allChecks [actorSig,nonzero amount,changes [nonce,(12,account (sub (o 12) amount) (add (o 12) amount)),(13,account (add (o 13) amount) (sub (o 13) amount))],noCash]
  changesAt src dst updates=allChecks [same (dst i) (maybe (src i) id (lookup i updates))|i<-[0..31]]

step state xs=update state ((14,state H.!!14+1):xs)
joinReference state who syWant ptWant=step state [(3+who,state H.!!(3+who)-sy),(5+who,state H.!!(5+who)-pt),(7,newSy),(8,newPt),(11,state H.!!11+mint+locked),(12+who,state H.!!(12+who)+mint),(18,newRate),(19,state H.!!19+locked)]
 where
  (sy,pt,mint,locked)=Q.joinInteger (state H.!!7) (state H.!!8) (state H.!!11) syWant ptWant
  newSy=state H.!!7+sy;newPt=state H.!!8+pt
  newRate=if state H.!!11==0 then P.initialRateInteger newSy newPt (state H.!!17) (P.year-state H.!!16) else state H.!!18
tradeReference state who buy quantity=(step state [(3+who,state H.!!(3+who)+net),(5+who,state H.!!(5+who)+(if buy then quantity else negate quantity)),(7,sy),(8,pt),(9,state H.!!9+reserve),(18,rate)],net,reserve)
 where
  (sy,pt,net,_,reserve,rate)=P.tradeInteger (state H.!!7) (state H.!!8) (state H.!!17) (P.year-state H.!!16) (state H.!!18) buy quantity
exitReference state who lp=step state [(3+who,state H.!!(3+who)+sy),(5+who,state H.!!(5+who)+pt),(7,state H.!!7-sy),(8,state H.!!8-pt),(11,state H.!!11-lp),(12+who,state H.!!(12+who)-lp)]
 where sy=lp*(state H.!!7) `div`(state H.!!11);pt=lp*(state H.!!8) `div`(state H.!!11)
